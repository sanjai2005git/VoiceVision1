# Cell 3: Data Parsing, Age Binning, and Preprocessing Setup
import os
import glob
import joblib
import pandas as pd
import numpy as np
import tensorflow as tf
from sklearn.model_selection import train_test_split
from sklearn.utils import class_weight

image_dir = "/content/indian-actor"
scalar_save_dir = "/content/drive/MyDrive/Deep_learning_project/Parental_Control_project/Sanjay_Upload/Age_Recognition/CNN-Models/Scalar"
os.makedirs(scalar_save_dir, exist_ok=True)

bin_labels = [f"{i}-{i+9}" for i in range(1, 60, 10)]  # ['1-10', '11-20', '21-30', '31-40', '41-50', '51-60']

def get_age_bin(age):
    for idx, i in enumerate(range(1, 60, 10)):
        if i <= age <= i + 9:
            return bin_labels[idx]
    return None  # Discard ages outside 1-60

# 2. Extract Filenames, Filter Out 61-100, and Parse Age
image_paths = glob.glob(os.path.join(extract_path, "**", "*.chip"), recursive=True)
if len(image_paths) == 0:
    image_paths = glob.glob(os.path.join(extract_path, "**", "*_*_*_*.jpg*"), recursive=True)

data = []
for path in image_paths:
    filename = os.path.basename(path)
    try:
        age = int(filename.split('_')[0])
        # Completely skip images aged outside 1 to 60 right at the start
        if age < 1 or age > 60:
            continue

        age_bin = get_age_bin(age)
        if age_bin is not None:
            data.append({'image_path': path, 'age': age, 'Class': age_bin})
    except Exception:
        continue

df = pd.DataFrame(data)

if len(df) == 0:
    raise ValueError(f"❌ No matching images found in {extract_path}.")

print(f"✅ Extracted and parsed {len(df)} images into {len(bin_labels)} classes (ages 1-60; 61+ removed).")

# Ensure categorical ordering across all 10 age bins
df['Class'] = pd.Categorical(df['Class'], categories=bin_labels)

# 3. Save Metadata/Class Mappings
metadata = {
    'bin_labels': bin_labels,
    'img_size': (224, 224)
}
joblib.dump(metadata, os.path.join(scalar_save_dir, "dataset_metadata.pkl"))
print(f"✅ Metadata saved to: {scalar_save_dir}")

# 4. Train/Test Split
train_df, val_df = train_test_split(df, test_size=0.20, random_state=42, stratify=df['Class'])

# Compute Class Weights
y_train_indices = train_df['Class'].cat.codes.values
computed_weights = class_weight.compute_class_weight('balanced', classes=np.unique(y_train_indices), y=y_train_indices)
class_weight_dict = dict(zip(np.unique(y_train_indices), computed_weights))

# 5. Data Generators
train_datagen = tf.keras.preprocessing.image.ImageDataGenerator(
    rotation_range=20,
    width_shift_range=0.1,
    height_shift_range=0.1,
    horizontal_flip=True,
    fill_mode='nearest'
)

val_datagen = tf.keras.preprocessing.image.ImageDataGenerator()

IMG_SIZE = (224, 224)
BATCH_SIZE = 32

train_generator = train_datagen.flow_from_dataframe(
    dataframe=train_df,
    x_col='image_path',
    y_col='Class',
    target_size=IMG_SIZE,
    batch_size=BATCH_SIZE,
    class_mode='categorical',
    shuffle=True
)

val_generator = val_datagen.flow_from_dataframe(
    dataframe=val_df,
    x_col='image_path',
    y_col='Class',
    target_size=IMG_SIZE,
    batch_size=BATCH_SIZE,
    class_mode='categorical',
    shuffle=False
)

# --- REPLACE ALL OF CELL 4 WITH THIS BLOCK ---
import tensorflow as tf
from tensorflow.keras import layers, models

# Enable Mixed Precision for GPU speedup
tf.keras.mixed_precision.set_global_policy('mixed_float16')

def build_convnext_tiny(num_classes=3):
    base_model = tf.keras.applications.ConvNeXtTiny(
        include_top=False,
        weights='imagenet',
        input_shape=(224, 224, 3)
    )

    base_model.trainable = True
    for layer in base_model.layers[:-30]:
        layer.trainable = False

    inputs = layers.Input(shape=(224, 224, 3))
    x = base_model(inputs)
    x = layers.GlobalAveragePooling2D()(x)
    x = layers.BatchNormalization()(x)
    x = layers.Dropout(0.3)(x)
    x = layers.Dense(256, activation='relu')(x)
    x = layers.Dropout(0.2)(x)

    # 3 output classes: ['1-20', '21-40', '41-60']
    outputs = layers.Dense(num_classes, activation='softmax', dtype='float32')(x)

    model = models.Model(inputs, outputs)
    return model

model = build_convnext_tiny(num_classes=6)

model.compile(
    optimizer=tf.keras.optimizers.Adam(learning_rate=1e-4),
    loss='categorical_crossentropy',
    metrics=['categorical_accuracy']
)

model.summary()

# Cell 5: Train, Save Best Model, and Plot Results across 10 Bins
import matplotlib.pyplot as plt
from datetime import datetime

model_save_dir = "/content/drive/MyDrive/Deep_learning_project/Parental_Control_project/Sanjay_Upload/Age_Recognition/CNN-Models"
os.makedirs(model_save_dir, exist_ok=True)

timestamp = datetime.now().strftime("%Y-%m-%d_%H-%M-%S")

    ReduceLROnPlateau(monitor='val_loss', factor=0.5, patience=3, min_lr=1e-7, verbose=1),
    EarlyStopping(monitor='val_loss', patience=7, restore_best_weights=True)

print("🚀 Starting Fast ConvNeXt Tiny Training on GPU...")
history = model.fit(
    train_generator,
    validation_data=val_generator,
    epochs=25,
    class_weight=class_weight_dict,
    callbacks=callbacks_list,
)

print(f"✅ Best model saved to: {model_save_path}")

# Cell 6: Individual Image Detection Test
import cv2
import glob
import numpy as np
import tensorflow as tf
import matplotlib.pyplot as plt

def predict_single_image(image_path, model_path):
    if not os.path.exists(image_path):
        print(f"❌ File not found: {image_path}")
        return

    model = tf.keras.models.load_model(model_path)
    bin_labels = [f"{i}-{i+9}" for i in range(1, 60, 10)]

    img = cv2.imread(image_path)
    img_rgb = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
    img_resized = cv2.resize(img_rgb, (224, 224))
    img_array = np.expand_dims(img_resized, axis=0)

    preds = model.predict(img_array, verbose=0)[0]
    predicted_class = bin_labels[np.argmax(preds)]
    confidence = np.max(preds) * 100

    # Extract true age from filename if present
    filename = os.path.basename(image_path)
    true_age = filename.split('_')[0] if '_' in filename else "Unknown"

    plt.figure(figsize=(6, 6))
    plt.imshow(img_rgb)
    plt.title(f"True Age: {true_age}\nPredicted Range: {predicted_class} ({confidence:.2f}%)", fontsize=12, color='green')
    plt.axis('off')
    plt.show()

    return predicted_class
